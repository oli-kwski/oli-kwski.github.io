{{- /* og-image.html Generates a 1200x630 OG card per page by overlaying the page's title onto assets/images/og-template.png at build time using Hugo's images.Text filter. Returns the resource so the caller can use .Permalink for meta tags (or .RelPermalink if you ever need a relative URL). Word wrap is greedy and based on a character-count heuristic tuned for Lato Black at 76px in a 1040px text box (~26 chars per line). Up to 3 lines render at full size; 4+ lines drop to 60px so they still fit. */ -}} {{- $tpl := resources.Get "images/og-template.png" -}} {{- $font := resources.Get "fonts/Lato-Black.ttf" -}} {{- $title := .Title | default .Site.Title -}} {{- /* --- Greedy line wrap by character count --- */ -}} {{- $maxChars := 26 -}} {{- $lines := slice -}} {{- $current := "" -}} {{- range (split $title " ") -}} {{- $candidate := cond (eq $current "") . (printf "%s %s" $current .) -}} {{- if le (len $candidate) $maxChars -}} {{- $current = $candidate -}} {{- else -}} {{- if ne $current "" -}} {{- $lines = $lines | append $current -}} {{- end -}} {{- $current = . -}} {{- end -}} {{- end -}} {{- if ne $current "" -}} {{- $lines = $lines | append $current -}} {{- end -}} {{- /* If wrapping at 26 chars yields more than 3 lines, redo at 32 chars and drop the font from 76px to 60px so the overflow fits. */ -}} {{- $size := 76 -}} {{- $linespacing := 18 -}} {{- if gt (len $lines) 3 -}} {{- $maxChars = 32 -}} {{- $size = 60 -}} {{- $linespacing = 14 -}} {{- $lines = slice -}} {{- $current = "" -}} {{- range (split $title " ") -}} {{- $candidate := cond (eq $current "") . (printf "%s %s" $current .) -}} {{- if le (len $candidate) $maxChars -}} {{- $current = $candidate -}} {{- else -}} {{- if ne $current "" -}} {{- $lines = $lines | append $current -}} {{- end -}} {{- $current = . -}} {{- end -}} {{- end -}} {{- if ne $current "" -}} {{- $lines = $lines | append $current -}} {{- end -}} {{- end -}} {{- /* Compose the final image. images.Text supports newlines + linespacing in a single Filter call, which is cheaper than chaining N filters. */ -}} {{- $titleText := delimit $lines "\n" -}} {{- $opts := dict "color" "#ffffff" "size" $size "linespacing" $linespacing "x" 80 "y" 100 "font" $font -}} {{- $img := $tpl.Filter (images.Text $titleText $opts) -}} {{- /* Re-encode as PNG so the cached resource has a stable path. */ -}} {{- return $img -}}
